% TABLEAU AUTOMATIQUE
%
% Le nombre de colonnes est calculé automatiquement
% à partir des headers.
% =========================================================

\ExplSyntaxOn

\seq_new:N \l_invoice_header_seq
\int_new:N \l_invoice_column_count_int

% Affichage automatique des headers avec leur style
\cs_new_protected:Npn \invoice_print_headers:
{
    \seq_map_indexed_inline:Nn \l_invoice_header_seq
    {
        \textbf{\color{white} ##2}

        \int_compare:nNnF
            {##1}
            =
            {\l_invoice_column_count_int}
            { & }
    }
}

% #1 = headers
% #2 = articles
\NewDocumentCommand{\InvoiceTable}{+m +m}
{
    % Découpe les headers à chaque &
    \seq_set_split:Nnn
        \l_invoice_header_seq
        { & }
        { #1 }

    % Compte automatiquement le nombre de colonnes
    \int_set:Nn
        \l_invoice_column_count_int
        { \seq_count:N \l_invoice_header_seq }

    \renewcommand{\arraystretch}{1.5}

    % Couleurs alternées des articles
    \rowcolors{2}{white}{lightgray}

    % Création automatique du bon nombre de colonnes X
    \begin{tabularx}
        {\textwidth}
        {*{\int_use:N \l_invoice_column_count_int}
        {>{\raggedright\arraybackslash}X}}

        % Première ligne = headers
        \rowcolor{green}
        \invoice_print_headers:
        \\

        % Articles
        #2

    \end{tabularx}

    \rowcolors{1}{}{}
}

\ExplSyntaxOff
